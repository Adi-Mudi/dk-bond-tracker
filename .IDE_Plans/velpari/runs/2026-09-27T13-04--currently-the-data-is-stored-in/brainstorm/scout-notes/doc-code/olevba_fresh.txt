Private Sub Worksheet_Change(ByVal Target As Range)
    ' This runs automatically whenever a cell is changed
    On Error Resume Next
    Application.ScreenUpdating = False
    
    Dim ws As Worksheet
    Set ws = Me
    Dim row As Long
    
    ' Check if change is in this sheet
    If Target.Parent.Name <> "Monthly Returns" Then
        Application.ScreenUpdating = True
        Exit Sub
    End If
    
    ' If change is in column H (Tenure) or column M (Starting Month), or month columns (N onwards)
    If Target.Column = 8 Or Target.Column = 13 Or Target.Column >= 14 Then
        row = Target.row
        
        ' Skip header row
        If row > 1 Then
            Call FormatRowPaymentTracking(ws, row)
        End If
    End If
    
    Application.ScreenUpdating = True
    On Error GoTo 0
End Sub

Sub FixBondDashboard()
    On Error GoTo ErrorHandler
    
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    
    Dim wsDashboard As Worksheet
    Dim wsMonthlyReturns As Worksheet
    
    ' 1. Check if Dashboard sheet exists
    On Error Resume Next
    Set wsDashboard = ThisWorkbook.Worksheets("Dashboard")
    If wsDashboard Is Nothing Then
        Set wsDashboard = ThisWorkbook.Worksheets.Add
        wsDashboard.Name = "Dashboard"
    End If
    On Error GoTo ErrorHandler
    
    ' 2. Check if Monthly Returns sheet exists
    On Error Resume Next
    Set wsMonthlyReturns = ThisWorkbook.Worksheets("Monthly Returns")
    If wsMonthlyReturns Is Nothing Then
        MsgBox "Monthly Returns sheet not found. Please ensure the sheet exists.", vbExclamation
        GoTo CleanUp
    End If
    On Error GoTo ErrorHandler
    
    ' 3. Run all updates
    Call ClearDashboard(wsDashboard)
    Call SetupDashboardHeaders(wsDashboard)
    Call UpdateDashboardSummary(wsDashboard, wsMonthlyReturns)
    Call UpdateBondsStatusOverview(wsDashboard, wsMonthlyReturns)
    Call CreateMonthlyHistoryTable(wsDashboard, wsMonthlyReturns) ' <--- The function causing error must be defined below
    Call ApplyDashboardFormatting(wsDashboard)
    
    MsgBox "Dashboard has been restored and updated successfully!", vbInformation
    
CleanUp:
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Exit Sub
    
ErrorHandler:
    MsgBox "Error in FixBondDashboard" & vbCrLf & _
           "Error Number: " & Err.Number & vbCrLf & _
           "Error Description: " & Err.Description, vbCritical
    Resume CleanUp
End Sub

' SUBROUTINE 3: UPDATE SUMMARY (Top Section)
' ---------------------------------------------------------
Sub UpdateDashboardSummary(wsDashboard As Worksheet, wsMonthlyReturns As Worksheet)
    Dim lastRow As Long
    Dim rupeeFmt As String, balanceFmt As String
    
    ' -- Variables for Monthly Calculation --
    Dim currentMonthStr As String
    Dim monthCol As Long
    Dim cellVal As Variant
    Dim monthColLetter As String
    Dim rngFound As Range
    
    rupeeFmt = "_([$" & ChrW(8377) & "] * #,##0.00_);_([$" & ChrW(8377) & "] * (#,##0.00);_([$" & ChrW(8377) & "] * ""-""??_);_(@_)"
    balanceFmt = "[$" & ChrW(8377) & "] #,##0.00;[Red][$" & ChrW(8377) & "] -#,##0.00;[$" & ChrW(8377) & "] 0.00"

    lastRow = wsMonthlyReturns.Cells(wsMonthlyReturns.Rows.Count, "A").End(xlUp).row
    If lastRow < 2 Then Exit Sub
    
    ' Standard Calculations
    wsDashboard.Range("B3").Formula = "=COUNTA('Monthly Returns'!E2:E" & lastRow & ")"
    wsDashboard.Range("B4").Formula = "=COUNTIF('Monthly Returns'!A2:A" & lastRow & ",""Danger"")"
    wsDashboard.Range("B5").Formula = "=COUNTIF('Monthly Returns'!A2:A" & lastRow & ",""On-going"")"
    wsDashboard.Range("B6").Formula = "=COUNTIF('Monthly Returns'!A2:A" & lastRow & ",""Completed"")"
    
    wsDashboard.Range("E3").Formula = "=SUM('Monthly Returns'!G2:G" & lastRow & ")"
    wsDashboard.Range("E4").Formula = "=SUM('Monthly Returns'!K2:K" & lastRow & ")"
    wsDashboard.Range("E5").Formula = "=SUM('Monthly Returns'!L2:L" & lastRow & ")"
    
    ' DYNAMIC "THIS MONTH RETURNS" CALCULATION
    currentMonthStr = Format(Date, "mmm-yy")
    monthCol = 0
    On Error Resume Next
        Set rngFound = wsMonthlyReturns.Rows(1).Find(What:=currentMonthStr, LookIn:=xlValues, LookAt:=xlWhole)
        If rngFound Is Nothing Then
            Dim c As Long
            For c = 14 To 100
                cellVal = wsMonthlyReturns.Cells(1, c).value
                If IsDate(cellVal) Then
                    If Format(cellVal, "mmm-yy") = currentMonthStr Then
                        Set rngFound = wsMonthlyReturns.Cells(1, c)
                        Exit For
                    End If
                End If
            Next c
        End If
    On Error GoTo 0
    
    If Not rngFound Is Nothing Then
        monthCol = rngFound.Column
        monthColLetter = Split(Cells(1, monthCol).Address, "$")(1)
        wsDashboard.Range("E6").Formula = "=SUM('Monthly Returns'!" & monthColLetter & "2:" & monthColLetter & lastRow & ")"
    Else
        wsDashboard.Range("E6").value = 0
    End If
    
    wsDashboard.Range("E3:E4").NumberFormat = rupeeFmt
    wsDashboard.Range("E6").NumberFormat = rupeeFmt
    wsDashboard.Range("E5").NumberFormat = balanceFmt
End Sub

' SUBROUTINE 4: CREATE MONTHLY HISTORY TABLE (The New Feature)
' ---------------------------------------------------------
Sub CreateMonthlyHistoryTable(wsDashboard As Worksheet, wsMonthlyReturns As Worksheet)
    ' CREATES A TABLE IN COLUMNS L & M SHOWING ALL MONTHLY RETURNS
    Dim lastCol As Long, r As Long, i As Long
    Dim monthDate As Variant
    Dim monthSum As Double
    Dim startCol As Long
    Dim lastDataRow As Long
    Dim rupeeFmt As String
    
    rupeeFmt = "_([$" & ChrW(8377) & "] * #,##0.00_);_([$" & ChrW(8377) & "] * (#,##0.00);_([$" & ChrW(8377) & "] * ""-""??_);_(@_)"
    lastDataRow = wsMonthlyReturns.Cells(wsMonthlyReturns.Rows.Count, "A").End(xlUp).row
    
    ' Header for History Table
    With wsDashboard.Range("L3:M3")
        .Merge
        .value = "FULL MONTHLY RETURNS HISTORY"
        .Font.Bold = True
        .Interior.Color = RGB(68, 114, 196)
        .Font.Color = RGB(255, 255, 255)
        .HorizontalAlignment = xlCenter
    End With
    
    wsDashboard.Range("L4").value = "Month"
    wsDashboard.Range("M4").value = "Total Returns"
    wsDashboard.Range("L4:M4").Font.Bold = True
    wsDashboard.Range("L4:M4").Interior.Color = RGB(217, 225, 242)
    wsDashboard.Range("L4:M4").Borders.LineStyle = xlContinuous
    
    ' Start scanning from Column N (14) in Monthly Returns
    startCol = 14
    lastCol = wsMonthlyReturns.Cells(1, wsMonthlyReturns.Columns.Count).End(xlToLeft).Column
    
    r = 5 ' Start output at Row 5
    
    For i = startCol To lastCol
        monthDate = wsMonthlyReturns.Cells(1, i).value
        
        ' Check if header is a valid date (to avoid empty columns)
        If IsDate(monthDate) Then
            ' Calculate Sum for this month column
            monthSum = Application.WorksheetFunction.Sum(wsMonthlyReturns.Range(wsMonthlyReturns.Cells(2, i), wsMonthlyReturns.Cells(lastDataRow, i)))
            
            ' Only list months that have valid data (or list all if you prefer)
            wsDashboard.Cells(r, 12).value = monthDate
            wsDashboard.Cells(r, 13).value = monthSum
            
            ' Highlight CURRENT MONTH
            If Format(monthDate, "mmm-yy") = Format(Date, "mmm-yy") Then
                wsDashboard.Range("L" & r & ":M" & r).Interior.Color = RGB(255, 255, 0) ' Yellow Highlight
            End If
            
            r = r + 1
        End If
    Next i
    
    ' Formatting History Table
    If r > 5 Then
        wsDashboard.Range("L5:L" & r - 1).NumberFormat = "mmm-yy"
        wsDashboard.Range("M5:M" & r - 1).NumberFormat = rupeeFmt
        wsDashboard.Range("L5:M" & r - 1).Borders.LineStyle = xlContinuous
        wsDashboard.Columns("L:M").AutoFit
    End If
End Sub

' SUBROUTINE 5: UPDATE MAIN TABLE
' ---------------------------------------------------------
Sub UpdateBondsStatusOverview(wsDashboard As Worksheet, wsMonthlyReturns As Worksheet)
    Dim lastRow As Long, i As Long, destRow As Long
    Dim statusPriority As Integer
    Dim bondData() As Variant
    Dim bondCount As Long
    Dim colMaturityDate As Long
    Dim rngHeader As Range
    Dim rupeeFmt As String, balanceFmt As String
    
    rupeeFmt = "_([$" & ChrW(8377) & "] * #,##0.00_);_([$" & ChrW(8377) & "] * (#,##0.00);_([$" & ChrW(8377) & "] * ""-""??_);_(@_)"
    balanceFmt = "[$" & ChrW(8377) & "] #,##0.00;[Red][$" & ChrW(8377) & "] -#,##0.00;[$" & ChrW(8377) & "] 0.00"

    lastRow = wsMonthlyReturns.Cells(wsMonthlyReturns.Rows.Count, "A").End(xlUp).row
    If lastRow < 2 Then Exit Sub
    
    ' Find Maturity Date Column
    On Error Resume Next
    Set rngHeader = wsMonthlyReturns.Rows(1).Find(What:="Maturity Date", LookIn:=xlValues, LookAt:=xlWhole)
    If Not rngHeader Is Nothing Then colMaturityDate = rngHeader.Column Else colMaturityDate = 9
    On Error GoTo 0

    wsDashboard.Range("A10:J1000").ClearContents
    bondCount = 0
    ReDim bondData(1 To lastRow - 1, 1 To 11)
    
    For i = 2 To lastRow
        If Trim(wsMonthlyReturns.Cells(i, 5).value) <> "" Then
            bondCount = bondCount + 1
            Select Case UCase(Trim(wsMonthlyReturns.Cells(i, 1).value))
                Case "DANGER": statusPriority = 1
                Case "ON-GOING": statusPriority = 2
                Case "COMPLETED": statusPriority = 3
                Case Else: statusPriority = 4
            End Select
            
            bondData(bondCount, 1) = wsMonthlyReturns.Cells(i, 1).value
            bondData(bondCount, 2) = wsMonthlyReturns.Cells(i, 6).value
            bondData(bondCount, 3) = wsMonthlyReturns.Cells(i, 4).value
            bondData(bondCount, 4) = wsMonthlyReturns.Cells(i, 5).value
            bondData(bondCount, 5) = wsMonthlyReturns.Cells(i, 7).value
            bondData(bondCount, 6) = wsMonthlyReturns.Cells(i, 8).value
            bondData(bondCount, 7) = wsMonthlyReturns.Cells(i, colMaturityDate).value
            bondData(bondCount, 8) = wsMonthlyReturns.Cells(i, 10).value
            bondData(bondCount, 9) = wsMonthlyReturns.Cells(i, 11).value
            bondData(bondCount, 10) = wsMonthlyReturns.Cells(i, 12).value
            bondData(bondCount, 11) = statusPriority
        End If
    Next i
    
    ' Bubble Sort
    If bondCount > 1 Then
        Dim temp As Variant, k As Long, swapped As Boolean
        Do
            swapped = False
            For i = 1 To bondCount - 1
                If bondData(i, 11) > bondData(i + 1, 11) Then
                    For k = 1 To 11
                        temp = bondData(i, k)
                        bondData(i, k) = bondData(i + 1, k)
                        bondData(i + 1, k) = temp
                    Next k
                    swapped = True
                End If
            Next i
        Loop While swapped
    End If
    
    destRow = 10
    For i = 1 To bondCount
        Dim statusRng As Range
        Set statusRng = wsDashboard.Cells(destRow, 1)
        statusRng.value = bondData(i, 1)
        
        Select Case UCase(Trim(bondData(i, 1)))
            Case "DANGER": statusRng.Interior.Color = RGB(255, 0, 0): statusRng.Font.Color = RGB(255, 255, 255): statusRng.Font.Bold = True
            Case "ON-GOING": statusRng.Interior.Color = RGB(255, 165, 0): statusRng.Font.Color = RGB(0, 0, 0): statusRng.Font.Bold = True
            Case "COMPLETED": statusRng.Interior.Color = RGB(0, 176, 80): statusRng.Font.Color = RGB(255, 255, 255): statusRng.Font.Bold = True
            Case Else: statusRng.Interior.ColorIndex = xlNone
        End Select
        
        wsDashboard.Cells(destRow, 2).Resize(1, 9).value = Array(bondData(i, 2), bondData(i, 3), bondData(i, 4), bondData(i, 5), bondData(i, 6), bondData(i, 7), bondData(i, 8), bondData(i, 9), bondData(i, 10))
        destRow = destRow + 1
    Next i
    
    If destRow > 10 Then
        wsDashboard.Range("E10:E" & destRow - 1).NumberFormat = rupeeFmt
        wsDashboard.Range("H10:I" & destRow - 1).NumberFormat = rupeeFmt
        wsDashboard.Range("J10:J" & destRow - 1).NumberFormat = balanceFmt
        wsDashboard.Range("G10:G" & destRow - 1).NumberFormat = "[$-en-US]DD-MMM-YYYY;@"
        wsDashboard.Range("F10:G" & destRow - 1).HorizontalAlignment = xlCenter
    End If
End Sub

' SUBROUTINE 6: APPLY FORMATTING

Sub ApplyDashboardFormatting(ws As Worksheet)
    ' Clean up merged cells
    On Error Resume Next
    ws.Range("A1:F1").UnMerge
    ws.Range("A8:J8").UnMerge
    On Error GoTo 0
    
    ' Row 1: Main Header (Blue)
    With ws.Range("A1:F1")
        .Merge: .HorizontalAlignment = xlCenter: .VerticalAlignment = xlCenter
        .Interior.Color = RGB(68, 114, 196): .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True: .Font.Size = 14
    End With
    
    ' Row 8: Section Header (Matches your image: Blue Bar)
    With ws.Range("A8:J8")
        .Merge: .HorizontalAlignment = xlCenter
        .Interior.Color = RGB(68, 114, 196)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .Borders.LineStyle = xlContinuous
    End With
    
    ' Borders for Data Table
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).row
    If lastRow >= 9 Then
        With ws.Range("A9:J" & lastRow)
            .Borders.LineStyle = xlContinuous: .Borders.Weight = xlThin
        End With
    End If
    
    ' Auto-fit
    ws.Columns("A:J").AutoFit
    ws.Columns("E:E").ColumnWidth = ws.Columns("E:E").ColumnWidth + 2
    ws.Columns("H:J").ColumnWidth = ws.Columns("H:J").ColumnWidth + 2
End Sub

Option Explicit

Sub InitializeMonthlyReturnsFormatting()
    On Error Resume Next
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("Monthly Returns")
    
    If Not ws Is Nothing Then
        Application.ScreenUpdating = False
        Application.Calculation = xlCalculationManual
        
        ' STEP 1: Regenerate Row 1 headers with correct dates
        Call RegenerateMonthlyHeaders(ws)
        
        ' STEP 2: Format everything
        Call FormatDateColumns(ws)
        Call ApplyFormattingToAllRows(ws)
        
        ' STEP 3: Automatically Insert Formulas for Columns K and L
        Call InjectFormulas(ws)
        
        Application.Calculation = xlCalculationAutomatic
        Application.ScreenUpdating = True
    End If
    On Error GoTo 0
End Sub

Sub InjectFormulas(ws As Worksheet)
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).row
    
    If lastRow < 2 Then Exit Sub
    
    ' Column K: Till Return Amount
    ' Formula: =SUM(N2:XFD2)  -- Sums everything from column N onwards
    ws.Range("K2:K" & lastRow).Formula = "=SUM(N2:XFD2)"
    
    ' Column L: Balance Amount
    ' Formula: =K2 - J2       -- (Till Return) - (Maturity Amount)
    ' Note: Your screenshot shows "K2 - J2" but typically balance is "Maturity - Paid" or "Paid - Maturity".
    ' Based on your screenshot showing negative balances (e.g., -52,990), K - J is correct.
    ws.Range("L2:L" & lastRow).Formula = "=K2-J2"
    
    ' Optional: Add currency formatting to these calculation columns
    ws.Range("K2:L" & lastRow).NumberFormat = "[$" & ChrW(8377) & "] #,##0.00;[Red][$" & ChrW(8377) & "] -#,##0.00"
End Sub

Sub RegenerateMonthlyHeaders(ws As Worksheet)
    ' Creates proper monthly headers in Row 1 starting from Column N
    ' Starting from Sep-2025 through Feb-2027 (18 months)
    Dim col As Long
    Dim i As Integer
    Dim headerDate As Date
    
    col = 14
    For i = 0 To 17
        headerDate = DateAdd("M", i, DateSerial(2025, 9, 1))
        ws.Cells(1, col).value = headerDate
        ws.Cells(1, col).NumberFormat = "mmm-yy"
        col = col + 1
    Next i
    
    With ws.Range(ws.Cells(1, 14), ws.Cells(1, 31))
        .HorizontalAlignment = xlCenter
        .Font.Bold = True
    End With
End Sub

Sub FormatDateColumns(ws As Worksheet)
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).row
    
    If lastRow > 1 Then
        ws.Range("B2:B" & lastRow).NumberFormat = "dd-mmm-yyyy"
        ws.Range("M2:M" & lastRow).NumberFormat = "dd-mmm-yyyy"
        
        Dim matCol As Range
        Set matCol = ws.Rows(1).Find("Maturity Date", LookIn:=xlValues, LookAt:=xlWhole)
        If Not matCol Is Nothing Then
             ws.Range(ws.Cells(2, matCol.Column), ws.Cells(lastRow, matCol.Column)).NumberFormat = "dd-mmm-yyyy"
        End If
    End If
End Sub

Sub ApplyFormattingToAllRows(ws As Worksheet)
    Dim lastRow As Long, i As Long
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).row
    
    For i = 2 To lastRow
        Call FormatRowPaymentTracking(ws, i)
    Next i
End Sub

Sub FormatRowPaymentTracking(ws As Worksheet, row As Long)
    Dim tenureVal As Variant, startingMonth As Variant
    Dim startColIndex As Long, endColIndex As Long
    Dim tenure As Long, col As Long
    Dim cellValue As Variant, rng As Range
    Dim rupeeFmt As String
    
    ' -- Variables for Status Logic --
    Dim paymentsMade As Long
    Dim status As String
    Dim lastMonthDate As Date
    Dim lastMonthCol As Long
    Dim isDanger As Boolean
    
    rupeeFmt = "[$" & ChrW(8377) & "] #,##0.00;[Red][$" & ChrW(8377) & "] -#,##0.00"
    
    tenureVal = ws.Cells(row, 8).value
    startingMonth = ws.Cells(row, 13).value
    
    ' Basic Validations
    If IsEmpty(tenureVal) Or Not IsNumeric(tenureVal) Then Exit Sub
    tenure = CLng(tenureVal)
    If tenure <= 0 Then Exit Sub
    If IsEmpty(startingMonth) Then Exit Sub
    
    startColIndex = FindStartingMonthColumn(ws, startingMonth)
    If startColIndex = -1 Then Exit Sub
    
    ' Clear Old Formatting
    Call ClearRowFormatting(ws, row)
    
    endColIndex = startColIndex + tenure - 1
    paymentsMade = 0
    
    ' Apply Colors & Count Payments
    For col = startColIndex To endColIndex
        If col <= 200 Then
            Set rng = ws.Cells(row, col)
            cellValue = rng.value
            rng.NumberFormat = rupeeFmt
            
            If cellValue = "" Or cellValue = 0 Or IsEmpty(cellValue) Then
                ' Payment Due (Empty)
                rng.Interior.Color = RGB(255, 200, 200) ' Light Red
                rng.Font.Color = RGB(0, 0, 0)
                rng.Font.Bold = False
            Else
                ' Payment Received
                rng.Interior.Color = RGB(200, 255, 200) ' Light Green
                rng.Font.Color = RGB(0, 100, 0)
                rng.Font.Bold = True
                paymentsMade = paymentsMade + 1
            End If
        End If
    Next col
    
    ' ---------------------------------------------------------
    ' AUTOMATIC STATUS LOGIC
    ' ---------------------------------------------------------
    status = "On-going" ' Default
    isDanger = False
    
    ' 1. Check for DANGER (Missed payment in LAST month only)
    ' Calculate Last Month (e.g. if Today is Dec 2025, Last Month is Nov 2025)
    lastMonthDate = DateSerial(Year(Date), Month(Date) - 1, 1)
    
    ' Find column for Last Month
    lastMonthCol = FindStartingMonthColumn(ws, lastMonthDate)
    
    ' Only check if Last Month is within this bond's active tenure range
    If lastMonthCol >= startColIndex And lastMonthCol <= endColIndex Then
        Dim lastMonthVal As Variant
        lastMonthVal = ws.Cells(row, lastMonthCol).value
        
        ' If Last Month cell is empty/0, mark as DANGER
        If IsEmpty(lastMonthVal) Or lastMonthVal = 0 Or lastMonthVal = "" Then
            status = "Danger"
            isDanger = True
        End If
    End If
    
    ' 2. Check for COMPLETED (All tenure payments made)
    If paymentsMade >= tenure Then
        status = "Completed"
    End If
    
    ' 3. Update Status Column (Column A)
    ' Only update if it's different to avoid infinite loops in Worksheet_Change
    If ws.Cells(row, 1).value <> status Then
        Application.EnableEvents = False ' Prevent triggering change event again
        ws.Cells(row, 1).value = status
        Application.EnableEvents = True
    End If
    
End Sub

Function FindStartingMonthColumn(ws As Worksheet, startingMonth As Variant) As Long
    Dim col As Long
    Dim headerVal As Variant
    Dim targetMonthYear As String
    Dim headerMonthYear As String
    Dim targetDate As Date
    Dim headerDate As Date
    
    targetDate = 0
    On Error Resume Next
    targetDate = CDate(startingMonth)
    On Error GoTo 0
    
    If targetDate = 0 Then
        FindStartingMonthColumn = -1
        Exit Function
    End If
    
    targetMonthYear = Format(targetDate, "mmm-yy")
    
    ' Scan columns N (14) to BZ (78)
    For col = 14 To 78
        headerVal = ws.Cells(1, col).value
        If Not IsEmpty(headerVal) Then
            headerDate = 0
            On Error Resume Next
            headerDate = CDate(headerVal)
            On Error GoTo 0
            
            If headerDate <> 0 Then
                headerMonthYear = Format(headerDate, "mmm-yy")
                If LCase(headerMonthYear) = LCase(targetMonthYear) Then
                    FindStartingMonthColumn = col
                    Exit Function
                End If
            End If
        End If
    Next col
    FindStartingMonthColumn = -1
End Function

Sub ClearRowFormatting(ws As Worksheet, row As Long)
    Dim rng As Range
    Set rng = ws.Range(ws.Cells(row, 14), ws.Cells(row, 100))
    rng.Interior.ColorIndex = xlNone
    rng.Font.Bold = False
    rng.Font.Color = RGB(0, 0, 0)
End Sub

Function IsNumeric(value As Variant) As Boolean
    On Error Resume Next
    IsNumeric = Not IsError(CDbl(value))
    On Error GoTo 0
End Function
